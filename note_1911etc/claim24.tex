\begin{exercise}

	Let $N \models \mathcal{T}$, ec for $\mathcal{T}^+$, such that $N \models \Phi(a)$. 
	Let $p(x) = pp-tp(a)$ (intended in the new language). Then $\mathcal{T} \cup p(x) \cup \{\exists y \phi(x,y) \}$ is consistent.
	As a consequence, $N \models \exists y \phi(a,y)$.
\end{exercise}

\begin{proof}

	By contradiction, let's say it is inconsistent. Then it is finitely inconsistent.
	Then $T \models \neg (\exists x \ \exists y \ \phi(x,y) \land \bigwedge_{k=1,..,n} \theta_k(x) )$ for some $\theta_1,..,\theta_n \in p(x)$.
	Conjunction of pp formulae is equivalent to a pp formula, so by definition of $\mathcal{T}^+$ we have 
	$\neg (\exists x \ \Phi(x) \land \bigwedge_{k=1,..,n} \theta_k(x) ) \in \mathcal{T}^+$.
	Then $N \models p(a) \land \Phi(a) \land \neg (\exists x \ \Phi(x) \land \bigwedge_{k=1,..,n} \theta_k(x))$, a contradiction.

	By compactness, we can find a model $M$ of $\mathcal{T}$ such that for some $c \in M^x$ we have $M \models p(c) \land \exists y \phi(c,y)$.
	Let $M'$ be an $|N|^+$-saturated elementary extension of $M$. 
	Then one can extend $a \mapsto c$ to a morphism from $N \to M'$.
	Since $M' \models \exists y \phi(c,y)$ and $N$ is ec (in the new language, therefore in the original) we get $N \models \exists y \phi(a,y)$.

\end{proof}




\begin{exercise}

	Let $N$ ec model fot $\mathcal{T}^+$ and assume $N \models \exists y \phi(a,y)$. Then $N \models \Phi(a)$.

\end{exercise}

\begin{proof}

	By contradiction, say $N \not \models \Phi(a)$. Now consider $N'$, whose semantic is the same as $N$ with the only difference that $N' \models \Phi(a)$.
	We claim that $N' \models \mathcal{T}^+$. 
	In fact otherwise there would be some sentence of the form $\rho \equiv \exists x_1,..,x_l \ \bigwedge_{k = 1,..,n} Q_k(\vec{x})$ such that
	$N' \models \rho$ but $\mathcal{T} \models \neg \rho'$, where $\rho'$ is obtained as the substituion of every occurrence in $\rho$ of $\Phi$ with $\exists y \phi$.
	Let $\vec{b}$ be the solution of $\rho$ in $N'$ and let $\vec{b_k}$ be what occurs in $Q_k$. Then split the istantion of $\rho$ as:
	\[
		\bigwedge_{k: Q_k \not = \Phi} Q_k(b_k) \land \bigwedge_{k: Q_k = \Phi, \vec{b_k} = \vec{a}} Q_k(a) \land \bigwedge_{k: Q_k = \Phi, \vec{b_k} \not = \vec{a}} Q_k(\vec{b_k})
	\]
	Now observe that 
	\[
		N \models \bigwedge_{k: Q_k \not = \Phi} Q_k(b_k) \land \bigwedge_{k: Q_k = \Phi, \vec{b_k} = \vec{a}} \exists y \phi(a,y) 
		\land \bigwedge_{k: Q_k = \Phi, \vec{b_k} \not = \vec{a}} Q_k(\vec{b_k})
	\]

	In fact:
	\begin{enumerate}
		\item	the first and third $\bigwedge$ are ok because $N,N'$ are the same here and $N'$ models it.
		\item	the second is ok because by hypothesis $N \models \exists y \phi(a,y)$.
	\end{enumerate}

	Finally, by the previous exercise we get
	\[
		N \models \bigwedge_{k: Q_k \not = \Phi} Q_k(b_k) \land \bigwedge_{k: Q_k = \Phi, \vec{b_k} = \vec{a}} \exists y \phi(a,y) \land 
		\bigwedge_{k: Q_k = \Phi, \vec{b_k} \not = \vec{a}} \exists y \phi(\vec{b_k},y)
	\]

	But this is a contradiction since this means that $N \models \rho$, so $N \not \models \mathcal{T}^+$.
	So far we proved that $N' \models \mathcal{T}^+$.
	Now observe that $N \to N', n \mapsto n$ is a morphism and $N' \models \Phi(a)$. Then using $N$ ec we get $N \models \Phi(a)$.

\end{proof}
